pub static UI_LISTENER: OnceLock<UnixListener> = OnceLock::new();

pub fn runtime_dir() -> Result<std::path::PathBuf> {
    let path = std::env::var_os("XDG_RUNTIME_DIR")
        .map(std::path::PathBuf::from)
        .ok_or_else(|| anyhow::anyhow!("Run inside your logged-in COSMIC desktop (XDG_RUNTIME_DIR is missing)"))?;
    Ok(path)
}

pub fn file_lock(name: &str, nonblocking: bool) -> Result<std::fs::File> {
    let file = std::fs::OpenOptions::new()
        .create(true)
        .truncate(false)
        .read(true)
        .write(true)
        .mode(0o600)
        .open(runtime_dir()?.join(name))?;
    let operation = libc::LOCK_EX | if nonblocking { libc::LOCK_NB } else { 0 };
    if unsafe { libc::flock(file.as_raw_fd(), operation) } != 0 {
        return Err(std::io::Error::last_os_error().into());
    }
    Ok(file)
}

pub struct SocketCleanup(pub std::path::PathBuf);
impl Drop for SocketCleanup {
    fn drop(&mut self) {
        let _ = std::fs::remove_file(&self.0);
    }
}

pub fn socket_path() -> std::path::PathBuf {
    runtime_dir()
        .expect("runtime checked in main")
        .join("cosmic-ext-app-switcher.sock")
}

#[derive(Parser, Debug)]
#[command(name = "cosmic-ext-app-switcher", version)]
pub struct Args {
    #[arg(long, default_value_t = false)]
    pub reverse: bool,
    /// Track window activation history continuously (run via systemd --user).
    #[arg(long)]
    pub daemon: bool,
    /// Check that the history service is reachable.
    #[arg(long)]
    pub mru_check: bool,
    /// Print current MRU order as JSON without opening the switcher.
    #[arg(long)]
    pub mru_list: bool,
    /// Report whether the compositor offers the Wayland interfaces the switcher needs, then exit
    #[arg(long, default_value_t = false)]
    pub check_compat: bool,
}

fn load_theme() -> Theme {
    use cosmic::cosmic_config::Config;
    Config::new(APP_ID, CONFIG_VERSION)
        .ok()
        .and_then(|c| c.get::<Theme>("theme").ok())
        .unwrap_or_default()
}

fn load_scope() -> WorkspaceScope {
    use cosmic::cosmic_config::Config;
    Config::new(APP_ID, CONFIG_VERSION)
        .ok()
        .and_then(|c| c.get::<WorkspaceScope>("workspace_scope").ok())
        .unwrap_or_default()
}

/// Prints one line per required Wayland global. Exits with status 1 if any is unusable.
fn check_compat() -> Result<()> {
    let statuses = wayland::probe_required_globals()?;
    for status in &statuses {
        println!("{status}");
    }
    if !statuses.iter().all(wayland::GlobalStatus::ok) {
        std::process::exit(1);
    }
    Ok(())
}

fn main() -> Result<()> {
    env_logger::init();
    let args = Args::parse();

    runtime_dir()?;
    if args.daemon {
        return wayland::run_daemon();
    }
    if args.mru_check {
        wayland::ping()?;
        println!("MRU service OK");
        return Ok(());
    }
    if args.mru_list {
        println!("{}", serde_json::to_string_pretty(&wayland::snapshot(load_scope())?)?);
        return Ok(());
    }
    if args.check_compat {
        return check_compat();
    }

    // Serialize startup so rapid shortcut invocations cannot open two overlays.
    let startup_lock = file_lock("cosmic-ext-app-switcher-ui.lock", false)?;
    let cmd = if args.reverse { b"prev" as &[u8] } else { b"next" as &[u8] };
    if let Ok(mut stream) = UnixStream::connect(socket_path()) {
        stream.set_write_timeout(Some(std::time::Duration::from_secs(1)))?;
        stream.write_all(cmd)?;
        return Ok(());
    }
    let _ = std::fs::remove_file(socket_path());
    let listener = UnixListener::bind(socket_path())?;
    std::fs::set_permissions(socket_path(), std::fs::Permissions::from_mode(0o600))?;
    listener.set_nonblocking(true)?;
    UI_LISTENER
        .set(listener)
        .map_err(|_| anyhow::anyhow!("UI socket already initialized"))?;
    let _cleanup = SocketCleanup(socket_path());

    let snapshot = wayland::snapshot(load_scope())?;
    drop(startup_lock);
    if snapshot.toplevels.is_empty() {
        return Ok(());
    }
    let res = app::run(snapshot.toplevels, args.reverse, snapshot.generation, load_theme());
    let _ = std::fs::remove_file(socket_path());
    res
}

pub struct AppSwitcher {
    core:          Core,
    pub toplevels: Vec<ToplevelEntry>,
    pub visuals:   Vec<AppVisual>,
    pub selected:  usize,
    generation:    String,
    pub finishing: bool,
    pub ready:     bool,
    super_held:    bool,
    alt_held:      bool,
    surface_id:    WindowId,
    pub theme:     ThemeValues,
}

#[derive(Debug, Clone)]
pub enum Message {
    SelectNext,
    SelectPrev,
    SelectIndex(usize),
    Activate,
    Cancel,
    Ready,
    KeyEvent(keyboard::Event),
}

impl Application for AppSwitcher {
    type Executor = cosmic::executor::Default;
    type Flags    = (Vec<ToplevelEntry>, bool, String, Theme);
    type Message  = Message;

    const APP_ID: &'static str = "io.github.cosmic-ext-app-switcher";

    fn core(&self) -> &Core { &self.core }
    fn core_mut(&mut self) -> &mut Core { &mut self.core }

    fn init(core: Core, flags: Self::Flags) -> (Self, Task<Message>) {
        let (toplevels, reverse, generation, theme_preset) = flags;
        let n = toplevels.len();
        let selected = if reverse {
            n.saturating_sub(1)
        } else {
            1.min(n.saturating_sub(1))
        };

        let theme = theme_preset.values();
        // Pre-resolve icons and labels once before surface creation so view() never hits disk
        let visuals: Vec<AppVisual> = toplevels
            .iter()
            .map(|entry| visual_for(&entry.app_id, &entry.title, theme.icon_size))
            .collect();

        let (surface_w, surface_h) = crate::ui::surface_size(toplevels.len(), &theme);

        let surface_id = WindowId::unique();
        let layer_task = get_layer_surface::<cosmic::Action<Message>>(SctkLayerSurfaceSettings {
            id: surface_id,
            layer: Layer::Overlay,
            keyboard_interactivity: KeyboardInteractivity::Exclusive,
            anchor: Anchor::empty(),
            size: Some((Some(surface_w), Some(surface_h))),
            ..Default::default()
        });

        // Wait 28ms for Wayland configure handshake & wgpu buffer stride stabilization
        // Also acts as a fast-switch grace period so sub-28ms Alt+Tab never flashes on screen
        let ready_task = Task::perform(
            async { tokio::time::sleep(Duration::from_millis(28)).await; },
            |_| cosmic::Action::App(Message::Ready),
        );

        (
            AppSwitcher {
                core,
                toplevels,
                visuals,
                selected,
                generation,
                finishing: false,
                ready: false,
                super_held: false,
                alt_held: true,
                surface_id,
                theme,
            },
            Task::batch([layer_task, ready_task]),
        )
    }

    fn update(&mut self, msg: Message) -> Task<Message> {
        if self.finishing { return Task::none(); }
        let n = self.toplevels.len();
        if n == 0 { return iced::exit(); }
        match msg {
            Message::Ready => {
                self.ready = true;
            }
            Message::SelectNext => {
                self.selected = (self.selected + 1) % n;
            }
            Message::SelectPrev => {
                self.selected = (self.selected + n - 1) % n;
            }
            Message::SelectIndex(i) => {
                if i < n { self.selected = i; }
            }
            Message::Activate => {
                self.finishing = true;
                self.ready = false;
                let _ = std::fs::remove_file(crate::socket_path());
                if let Err(error) = crate::wayland::activate(
                    &self.generation, self.toplevels[self.selected].handle_key,
                ) {
                    log::error!("Could not activate selected window: {error:#}");
                }
                return Task::batch([
                    destroy_layer_surface(self.surface_id),
                    iced::exit(),
                ]);
            }
            Message::Cancel => {
                self.finishing = true;
                self.ready = false;
                let _ = std::fs::remove_file(crate::socket_path());
                return Task::batch([
                    destroy_layer_surface(self.surface_id),
                    iced::exit(),
                ]);
            }
            Message::KeyEvent(ke) => match ke {
                keyboard::Event::KeyPressed { key, modifiers, .. } => match key {
                    iced::keyboard::Key::Named(Named::Tab) => {
                        return self.update(if modifiers.shift() {
                            Message::SelectPrev
                        } else {
                            Message::SelectNext
                        });
                    }
                    iced::keyboard::Key::Named(Named::ArrowRight) | iced::keyboard::Key::Named(Named::ArrowDown) => {
                        return self.update(Message::SelectNext);
                    }
                    iced::keyboard::Key::Named(Named::ArrowLeft) | iced::keyboard::Key::Named(Named::ArrowUp) => {
                        return self.update(Message::SelectPrev);
                    }
                    iced::keyboard::Key::Named(Named::Escape) => {
                        return self.update(Message::Cancel);
                    }
                    iced::keyboard::Key::Named(Named::Enter) => {
                        return self.update(Message::Activate);
                    }
                    _ => {}
                },
                keyboard::Event::KeyReleased { key, .. } => match key {
                    iced::keyboard::Key::Named(Named::Alt)
                    | iced::keyboard::Key::Named(Named::AltGraph)
                    | iced::keyboard::Key::Named(Named::Super) => {
                        return self.update(Message::Activate);
                    }
                    _ => {}
                },
                keyboard::Event::ModifiersChanged(mods) => {
                    if !mods.alt() && !mods.logo() {
                        return self.update(Message::Activate);
                    }
                    self.super_held = mods.logo();
                    self.alt_held = mods.alt();
                }
            },
        }
        Task::none()
    }

    fn subscription(&self) -> Subscription<Message> {
        let key_sub = event::listen_with(|event, _status, _window| {
            match event {
                Event::Keyboard(ke) => Some(Message::KeyEvent(ke)),
                Event::Window(iced::window::Event::Unfocused) => Some(Message::Cancel),
                _ => None,
            }
        });

        let socket_sub = {
            use std::any::TypeId;
            use cosmic::iced::futures::SinkExt;
            struct SocketSub;
            Subscription::run_with(TypeId::of::<SocketSub>(), |_| {
                cosmic::iced::stream::channel(16, |mut tx: cosmic::iced::futures::channel::mpsc::Sender<Message>| async move {
                    use tokio::io::AsyncReadExt;
                    let listener = tokio::net::UnixListener::from_std(
                        crate::UI_LISTENER.get().expect("UI socket initialized").try_clone().expect("clone UI socket")
                    ).expect("nonblocking UI socket");
                    loop {
                        if let Ok((mut stream, _)) = listener.accept().await {
                            let mut buf = Vec::new();
                            let read = tokio::time::timeout(std::time::Duration::from_secs(1),
                                (&mut stream).take(16).read_to_end(&mut buf)).await;
                            if !matches!(read, Ok(Ok(_))) { continue; }
                            let msg = match buf.as_slice() {
                                b"next" => Message::SelectNext,
                                b"prev" => Message::SelectPrev,
                                _ => continue,
                            };
                            let _ = tx.send(msg).await;
                        }
                    }
                })
            })
        };

        Subscription::batch([key_sub, socket_sub])
    }

    fn view(&self) -> Element<'_, Message> {
        crate::ui::view(self)
    }

    fn view_window(&self, _id: WindowId) -> Element<'_, Message> {
        crate::ui::view(self)
    }
}

pub fn run(
    toplevels: Vec<ToplevelEntry>,
    reverse: bool,
    generation: String,
    theme: Theme,
) -> Result<()> {
    let settings = Settings::default()
        .no_main_window(true)
        .transparent(true);
    cosmic::app::run::<AppSwitcher>(settings, (toplevels, reverse, generation, theme))
        .map_err(|e| anyhow::anyhow!("{e:?}"))
}



use cosmic::widget::icon;
use freedesktop_desktop_entry::{DesktopEntry, Iter as DesktopIter};
use std::collections::HashSet;
use std::path::{Path, PathBuf};

const FALLBACK_ICON: &str = "application-x-executable";

pub enum AppIcon {
    Named(String),
    File(PathBuf),
}

pub struct AppVisual {
    pub icon: AppIcon,
    pub label: String,
}

pub fn visual_for(app_id: &str, window_title: &str, size: u16) -> AppVisual {
    if app_id.starts_with("steam_app_") {
        return AppVisual {
            icon: resolve("steam", size),
            label: first_non_empty(&[window_title, "Steam"]).to_string(),
        };
    }

    let entry = desktop_entry_for(app_id);
    let label = entry.as_ref()
        .and_then(|e| e.name.as_deref())
        .filter(|n| !n.is_empty())
        .unwrap_or_else(|| first_non_empty(&[window_title, app_id, "Unknown"]))
        .to_string();

    let icon = match entry.and_then(|e| e.icon) {
        Some(field) if Path::new(&field).is_absolute() && Path::new(&field).is_file() => {
            AppIcon::File(PathBuf::from(field))
        }
        Some(field) => resolve(&icon_name_from_field(&field), size),
        None => resolve(app_id.split('.').next_back().unwrap_or(app_id), size),
    };

    AppVisual { icon, label }
}

fn first_non_empty<'a>(candidates: &[&'a str]) -> &'a str {
    candidates.iter().copied().find(|c| !c.is_empty()).unwrap_or_default()
}

fn resolve(name: &str, size: u16) -> AppIcon {
    if icon::from_name(name).size(size).path().is_some() {
        return AppIcon::Named(name.to_string());
    }
    log::debug!("icon {name:?} not found at size {size}, falling back to {FALLBACK_ICON}");
    AppIcon::Named(FALLBACK_ICON.to_string())
}

struct DesktopInfo {
    icon: Option<String>,
    name: Option<String>,
}

fn desktop_entry_for(app_id: &str) -> Option<DesktopInfo> {
    if app_id.is_empty() {
        return None;
    }
    let dirs = search_dirs();

    for dir in &dirs {
        if let Some(info) = read_entry(&dir.join(format!("{app_id}.desktop"))) {
            return Some(info);
        }
    }

    for path in DesktopIter::new(dirs.clone()) {
        let Ok(bytes) = std::fs::read(&path) else { continue };
        let Ok(s) = std::str::from_utf8(&bytes) else { continue };
        let Ok(entry) = DesktopEntry::decode(&path, s) else { continue };
        let stem = path.file_stem().and_then(|s| s.to_str()).unwrap_or_default();
        let claims = entry.startup_wm_class().is_some_and(|c| c.eq_ignore_ascii_case(app_id))
            || stem_matches(stem, app_id);
        if claims {
            return Some(DesktopInfo {
                icon: entry.icon().map(ToString::to_string),
                name: entry.name(None).map(|n| n.to_string()),
            });
        }
    }

    log::debug!("no .desktop entry found for app_id {app_id:?}");
    None
}

fn stem_matches(stem: &str, app_id: &str) -> bool {
    stem.eq_ignore_ascii_case(app_id)
        || stem.rsplit('.').next().is_some_and(|last| last.eq_ignore_ascii_case(app_id))
}

fn read_entry(path: &Path) -> Option<DesktopInfo> {
    let bytes = std::fs::read(path).ok()?;
    let s = std::str::from_utf8(&bytes).ok()?;
    let entry = DesktopEntry::decode(path, s).ok()?;
    Some(DesktopInfo {
        icon: entry.icon().map(ToString::to_string),
        name: entry.name(None).map(|n| n.to_string()),
    })
}

fn search_dirs() -> Vec<PathBuf> {
    let home = std::env::var("HOME").unwrap_or_default();
    let data_home = std::env::var("XDG_DATA_HOME")
        .ok()
        .filter(|d| !d.is_empty())
        .unwrap_or_else(|| format!("{home}/.local/share"));
    let data_dirs = std::env::var("XDG_DATA_DIRS")
        .ok()
        .filter(|d| !d.is_empty())
        .unwrap_or_else(|| "/usr/local/share:/usr/share".to_string());

    let xdg = std::iter::once(data_home.clone())
        .chain(data_dirs.split(':').map(ToString::to_string))
        .filter(|d| !d.is_empty())
        .map(|d| PathBuf::from(d).join("applications"));

    let extra = [
        PathBuf::from("/var/lib/flatpak/exports/share/applications"),
        PathBuf::from(format!("{data_home}/flatpak/exports/share/applications")),
        PathBuf::from("/var/lib/snapd/desktop/applications"),
    ];

    let mut seen = HashSet::new();
    xdg.chain(extra).filter(|d| seen.insert(d.clone())).collect()
}

fn icon_name_from_field(icon: &str) -> String {
    if icon.starts_with('/') {
        Path::new(icon)
            .file_stem()
            .and_then(|s| s.to_str())
            .unwrap_or(icon)
            .to_string()
    } else {
        icon.to_string()
    }
}

#[cfg(test)]
mod tests {
    use super::{
        AppIcon, FALLBACK_ICON, icon_name_from_field, search_dirs, stem_matches, visual_for,
    };
    use std::path::PathBuf;

    #[test]
    fn absolute_icon_fields_become_stems() {
        assert_eq!(icon_name_from_field("/a/b/steam_icon_480.png"), "steam_icon_480");
        assert_eq!(icon_name_from_field("firefox"), "firefox");
    }

    #[test]
    fn stems_match_case_insensitively_and_by_last_component() {
        assert!(stem_matches("Firefox", "firefox"));
        assert!(stem_matches("org.jdownloader.JDownloader", "jdownloader"));
        assert!(!stem_matches("org.gnome.Nautilus", "gnome"));
        assert!(!stem_matches("firefox", "chromium"));
    }

    #[test]
    fn resolution_against_a_fixture_session() {
        let root = std::env::temp_dir().join(format!("switcher-icons-{}", std::process::id()));
        let apps = root.join("data_home/applications");
        let icon = root.join("thing.png");
        std::fs::create_dir_all(&apps).unwrap();
        std::fs::write(&icon, b"not really a png").unwrap();
        std::fs::write(
            apps.join("org.example.Thing.desktop"),
            format!("[Desktop Entry]\nType=Application\nName=Thing\nIcon={}\n", icon.display()),
        )
        .unwrap();

        unsafe {
            std::env::set_var("HOME", &root);
            std::env::set_var("XDG_DATA_HOME", root.join("data_home"));
            std::env::set_var("XDG_DATA_DIRS", "/usr/share");
        }

        let dirs = search_dirs();
        for expected in [
            PathBuf::from("/usr/share/applications"),
            apps.clone(),
            PathBuf::from("/var/lib/flatpak/exports/share/applications"),
            root.join("data_home/flatpak/exports/share/applications"),
            PathBuf::from("/var/lib/snapd/desktop/applications"),
        ] {
            assert!(dirs.contains(&expected), "missing {} in {dirs:?}", expected.display());
        }
        let mut deduped = dirs.clone();
        deduped.sort();
        deduped.dedup();
        assert_eq!(deduped.len(), dirs.len(), "duplicate dirs in {dirs:?}");

        for app_id in ["org.example.Thing", "thing"] {
            let visual = visual_for(app_id, "Some Window Title", 48);
            match visual.icon {
                AppIcon::File(p) => assert_eq!(p, icon, "for app_id {app_id:?}"),
                AppIcon::Named(n) => panic!("app_id {app_id:?} resolved to name {n:?}"),
            }
            assert_eq!(visual.label, "Thing", "for app_id {app_id:?}");
        }

        let visual = visual_for("no-such-app-6f3b", "Some Window Title", 48);
        match visual.icon {
            AppIcon::Named(n) => assert_eq!(n, FALLBACK_ICON),
            AppIcon::File(p) => panic!("unknown app_id resolved to file {}", p.display()),
        }
        assert_eq!(visual.label, "Some Window Title");
        assert_eq!(visual_for("no-such-app-6f3b", "", 48).label, "no-such-app-6f3b");

        std::fs::remove_dir_all(&root).ok();
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ToplevelEntry {
    pub app_id:     String,
    pub title:      String,
    pub is_active:  bool,
    pub handle_key: u64,
}

#[derive(Serialize, Deserialize)]
pub struct Snapshot {
    pub generation: String,
    pub toplevels: Vec<ToplevelEntry>,
}

#[derive(Serialize, Deserialize)]
enum Request {
    List(WorkspaceScope),
    Activate { generation: String, key: u64 },
    Ping,
}

#[derive(Serialize, Deserialize)]
enum Response {
    List(Snapshot),
    Ok,
    Error(String),
}

pub fn daemon_socket_path() -> Result<std::path::PathBuf> {
    Ok(crate::runtime_dir()?.join("cosmic-ext-app-switcher-mru.sock"))
}

fn request(request: &Request) -> Result<Response> {
    let mut stream = UnixStream::connect(daemon_socket_path()?)
        .context("MRU service is not running. Run: systemctl --user restart cosmic-ext-app-switcher-mru.service")?;
    stream.set_read_timeout(Some(Duration::from_secs(2)))?;
    stream.set_write_timeout(Some(Duration::from_secs(2)))?;
    serde_json::to_writer(&mut stream, request)?;
    stream.shutdown(std::net::Shutdown::Write)?;
    let mut bytes = Vec::new();
    stream.take(4 * 1024 * 1024).read_to_end(&mut bytes)?;
    let reply: Response = serde_json::from_slice(&bytes)?;
    if let Response::Error(message) = &reply {
        return Err(anyhow!("{message}"));
    }
    Ok(reply)
}

pub fn snapshot(scope: WorkspaceScope) -> Result<Snapshot> {
    match request(&Request::List(scope))? {
        Response::List(snapshot) => Ok(snapshot),
        _ => Err(anyhow!("Unexpected MRU service response")),
    }
}

pub fn activate(generation: &str, key: u64) -> Result<()> {
    match request(&Request::Activate { generation: generation.into(), key })? {
        Response::Ok => Ok(()),
        _ => Err(anyhow!("Unexpected MRU activation response")),
    }
}

pub fn ping() -> Result<()> {
    match request(&Request::Ping)? {
        Response::Ok => Ok(()),
        _ => Err(anyhow!("Unexpected MRU health response")),
    }
}

const TOPLEVEL_INFO_VERSION:    u32 = 1;
const TOPLEVEL_MANAGER_VERSION: u32 = 1;
const SEAT_VERSION:             u32 = 7;

const REQUIRED_GLOBALS: &[(&str, u32)] = &[
    ("zcosmic_toplevel_info_v1",    TOPLEVEL_INFO_VERSION),
    ("zcosmic_toplevel_manager_v1", TOPLEVEL_MANAGER_VERSION),
    ("zwlr_layer_shell_v1",         1),
    ("wl_seat",                     SEAT_VERSION),
];

pub struct GlobalStatus {
    interface:  &'static str,
    needed:     u32,
    advertised: Option<u32>,
}

impl GlobalStatus {
    pub fn ok(&self) -> bool {
        self.advertised.is_some_and(|v| v >= self.needed)
    }
}

impl std::fmt::Display for GlobalStatus {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        let (interface, needed) = (self.interface, self.needed);
        match self.advertised {
            Some(v) if v >= needed => write!(f, "{interface}: ok (v{v}, needs v{needed})"),
            Some(v) => write!(f, "{interface}: TOO OLD (v{v}, needs v{needed})"),
            None    => write!(f, "{interface}: NOT offered (needs v{needed})"),
        }
    }
}

fn required_globals_status(advertised: &[(String, u32)]) -> Vec<GlobalStatus> {
    REQUIRED_GLOBALS.iter()
        .map(|&(interface, needed)| GlobalStatus {
            interface,
            needed,
            advertised: advertised.iter()
                .filter(|(i, _)| i == interface)
                .map(|&(_, v)| v)
                .max(),
        })
        .collect()
}

pub fn probe_required_globals() -> Result<Vec<GlobalStatus>> {
    let conn = Connection::connect_to_env()
        .context("could not connect to the Wayland compositor (is COSMIC running?)")?;
    let (globals, _queue) = registry_queue_init::<RegistryProbe>(&conn)?;
    let advertised: Vec<(String, u32)> = globals.contents().clone_list()
        .into_iter()
        .map(|g| (g.interface, g.version))
        .collect();
    Ok(required_globals_status(&advertised))
}

struct Toplevel {
    key: u64,
    handle:     ZcosmicToplevelHandleV1,

    is_active:  bool,

pub fn run_daemon() -> Result<()> {
    let _lock = crate::file_lock("cosmic-ext-app-switcher-mru.lock", true)?;
    let conn    = Connection::connect_to_env()?;
    let display = conn.display();
    let mut queue = conn.new_event_queue();
    let qh = queue.handle();

    let registry = display.get_registry(&qh, ());

    let mut data = AppData {
        history: History::default(),
        next_key: 0,
        generation: format!("{}-{}", std::process::id(), SystemTime::now().duration_since(UNIX_EPOCH)?.as_nanos()),
        toplevels: vec![],
        _info: None,
        manager: None,
        seat: None,
        _workspace_manager: None,
        outputs: vec![],
        toplevel_info_name: None,
        globals: vec![],
    };

    queue.roundtrip(&mut data)?;

    let missing: Vec<String> = required_globals_status(&data.globals).iter()
        .filter(|s| !s.ok())
        .map(ToString::to_string)
        .collect();
    if !missing.is_empty() {
        return Err(anyhow!(
            "cosmic-ext-app-switcher can't run: the compositor doesn't offer the Wayland \
             interfaces it needs ({}). Run `cosmic-ext-app-switcher --check-compat` for details.",
            missing.join("; "),
        ));
    }

    queue.roundtrip(&mut data)?;

    if let Some(name) = data.toplevel_info_name.take() {
        data._info = Some(
            registry.bind::<ZcosmicToplevelInfoV1, _, _>(name, TOPLEVEL_INFO_VERSION, &qh, ())
        );
    }

    for _ in 0..5 {
        queue.roundtrip(&mut data)?;
    }

    let path = daemon_socket_path()?;
    let _ = std::fs::remove_file(&path);
    let listener = UnixListener::bind(&path)?;
    std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o600))?;
    listener.set_nonblocking(true)?;
    let _cleanup = crate::SocketCleanup(path);
    log::info!("MRU service ready; history starts now");

    loop {
        queue.dispatch_pending(&mut data)?;
        conn.flush()?;
        let Some(guard) = queue.prepare_read() else { continue; };
        let mut fds = [
            libc::pollfd { fd: conn.backend().poll_fd().as_raw_fd(), events: libc::POLLIN, revents: 0 },
            libc::pollfd { fd: listener.as_raw_fd(), events: libc::POLLIN, revents: 0 },
        ];
        let ready = unsafe { libc::poll(fds.as_mut_ptr(), fds.len() as libc::nfds_t, -1) };
        if ready < 0 {
            drop(guard);
            let error = std::io::Error::last_os_error();
            if error.kind() == std::io::ErrorKind::Interrupted { continue; }
            return Err(error.into());
        }
        if fds[0].revents != 0 {
            guard.read()?;
            queue.dispatch_pending(&mut data)?;
        } else {
            drop(guard);
        }
        if fds[1].revents & libc::POLLIN != 0 {
            let (mut stream, _) = listener.accept()?;
            stream.set_read_timeout(Some(Duration::from_millis(500)))?;
            stream.set_write_timeout(Some(Duration::from_millis(500)))?;
            let mut bytes = Vec::new();
            if (&mut stream).take(65536).read_to_end(&mut bytes).is_err() { continue; }
            let reply = match serde_json::from_slice::<Request>(&bytes) {
                Ok(Request::List(scope)) => {
                    queue.roundtrip(&mut data)?;
                    Response::List(data.snapshot(scope))
                }
                Ok(Request::Activate { generation, key }) => {
                    queue.roundtrip(&mut data)?;
                    if generation != data.generation {
                        Response::Error("MRU service restarted; reopen Alt+Tab".into())
                    } else if let Some(t) = data.toplevels.iter().find(|t| t.key == key) {
                        if let (Some(manager), Some(seat)) = (&data.manager, &data.seat) {
                            manager.activate(&t.handle, seat);
                            queue.roundtrip(&mut data)?;
                            Response::Ok
                        } else {
                            Response::Error("Activation protocol unavailable".into())
                        }
                    } else {
                        Response::Error("Selected window has already closed".into())
                    }
                }
                Ok(Request::Ping) => Response::Ok,
                Err(error) => Response::Error(error.to_string()),
            };
            let _ = serde_json::to_writer(&mut stream, &reply);
            let _ = stream.flush();
        }
    }
}

impl AppData {
    fn snapshot(&self, scope: WorkspaceScope) -> Snapshot {
        let reference = self.toplevels.iter()
            .filter(|t| !t.app_id.is_empty() || !t.title.is_empty())
            .min_by_key(|t| self.history.sort_key(t.key));
        let active_outputs = reference.map(|t| t.outputs.as_slice()).unwrap_or(&[]);
        let mut entries: Vec<_> = self.toplevels.iter()
            .filter(|t| !t.app_id.is_empty() || !t.title.is_empty())
            .filter(|t| match scope {
                WorkspaceScope::AllWorkspaces => true,
                WorkspaceScope::CurrentWorkspace => true,
                WorkspaceScope::CurrentOutput => active_outputs.is_empty()
                    || t.outputs.iter().any(|o| active_outputs.contains(o)),
            })
            .map(|t| ToplevelEntry {
                app_id: t.app_id.clone(),
                title: t.title.clone(),
                is_active: t.is_active,
                handle_key: t.key,
            }).collect();
        entries.sort_by_key(|t| self.history.sort_key(t.handle_key));
        Snapshot { generation: self.generation.clone(), toplevels: entries }
    }
}

impl Dispatch<wl_registry::WlRegistry, ()> for AppData {
    fn event(
        data: &mut Self,
        registry: &wl_registry::WlRegistry,
        event: wl_registry::Event,
        _: &(),
        _: &Connection,
        qh: &QueueHandle<Self>,
    ) {
        if let wl_registry::Event::Global { name, interface, version } = event {
            data.globals.push((interface.clone(), version));
            match interface.as_str() {
                "zcosmic_toplevel_info_v1" if version >= TOPLEVEL_INFO_VERSION => {
                    data.toplevel_info_name = Some(name);
                }
                "zcosmic_toplevel_manager_v1" if version >= TOPLEVEL_MANAGER_VERSION => {
                    data.manager = Some(registry.bind::<ZcosmicToplevelManagerV1, _, _>(
                        name, TOPLEVEL_MANAGER_VERSION, qh, (),
                    ));
                }
                "wl_seat" if version >= SEAT_VERSION => {
                    if data.seat.is_none() {
                        data.seat = Some(
                            registry.bind::<wl_seat::WlSeat, _, _>(name, SEAT_VERSION, qh, ())
                        );
                    }
                }
                "wl_output" => {
                    data.outputs.push(
                        registry.bind::<wl_output::WlOutput, _, _>(name, 1, qh, ())
                    );
                }
                "zcosmic_workspace_manager_v1" => {
                    data._workspace_manager = Some(
                        registry.bind::<ZcosmicWorkspaceManagerV1, _, _>(name, 1, qh, ())
                    );
                }
                _ => {}
            }
        }
    }
}

impl Dispatch<ZcosmicToplevelInfoV1, ()> for AppData {
    fn event(
        data: &mut Self,
        _: &ZcosmicToplevelInfoV1,
        event: zcosmic_toplevel_info_v1::Event,
        _: &(),
        _: &Connection,
        _: &QueueHandle<Self>,
    ) {
        if let zcosmic_toplevel_info_v1::Event::Toplevel { toplevel } = event {
            data.next_key += 1;
            data.history.open(data.next_key);
            data.toplevels.push(Toplevel {
                key: data.next_key,
                handle:     toplevel,
                app_id:     String::new(),
                title:      String::new(),
                is_active:  false,
                outputs:    vec![],
                workspaces: vec![],
            });
        }
    }

    event_created_child!(AppData, ZcosmicToplevelInfoV1, [
        zcosmic_toplevel_info_v1::EVT_TOPLEVEL_OPCODE =>
            (ZcosmicToplevelHandleV1, ()),
    ]);
}

impl Dispatch<ZcosmicToplevelHandleV1, ()> for AppData {
    fn event(
        data: &mut Self,
        handle: &ZcosmicToplevelHandleV1,
        event: zcosmic_toplevel_handle_v1::Event,
        _: &(),
        _: &Connection,
        _: &QueueHandle<Self>,
    ) {
        let Some(t) = data.toplevels.iter_mut().find(|t| &t.handle == handle) else {
            return;
        };
        match event {
            zcosmic_toplevel_handle_v1::Event::AppId { app_id } => {
                t.app_id = app_id;
            }
            zcosmic_toplevel_handle_v1::Event::Title { title } => {
                t.title = title;
            }
            zcosmic_toplevel_handle_v1::Event::State { state } => {
                let activated = zcosmic_toplevel_handle_v1::State::Activated as u32;
                t.is_active = state
                    .chunks_exact(4)
                    .any(|b| u32::from_ne_bytes(b.try_into().unwrap()) == activated);
                data.history.state(t.key, t.is_active);
            }
            zcosmic_toplevel_handle_v1::Event::Closed => {
                data.history.close(t.key);
                data.toplevels.retain(|t| &t.handle != handle);
            }
            zcosmic_toplevel_handle_v1::Event::OutputEnter { output } => {
                if !t.outputs.contains(&output) {
                    t.outputs.push(output);
                }
            }
            zcosmic_toplevel_handle_v1::Event::OutputLeave { output } => {
                t.outputs.retain(|o| o != &output);
            }
            zcosmic_toplevel_handle_v1::Event::WorkspaceEnter { workspace } => {
                if !t.workspaces.contains(&workspace) {
                    t.workspaces.push(workspace);
                }
            }
            zcosmic_toplevel_handle_v1::Event::WorkspaceLeave { workspace } => {
                t.workspaces.retain(|w| w != &workspace);
            }
            _ => {}
        }
    }
}

#[cfg(test)]
mod tests {
    use super::required_globals_status;

    fn advertised(globals: &[(&str, u32)]) -> Vec<(String, u32)> {
        globals.iter().map(|&(i, v)| (i.to_string(), v)).collect()
    }

    fn line_for(globals: &[(&str, u32)], interface: &str) -> String {
        required_globals_status(&advertised(globals)).iter()
            .map(ToString::to_string)
            .find(|l| l.starts_with(&format!("{interface}:")))
            .unwrap()
    }

    #[test]
    fn everything_offered_is_ok() {
        let globals = [
            ("zcosmic_toplevel_info_v1", 3),
            ("zcosmic_toplevel_manager_v1", 4),
            ("zwlr_layer_shell_v1", 5),
            ("wl_seat", 9),
        ];
        assert!(required_globals_status(&advertised(&globals)).iter().all(|s| s.ok()));
    }

    #[test]
    fn missing_global_is_reported() {
        let globals = [("zcosmic_toplevel_info_v1", 3), ("zwlr_layer_shell_v1", 5), ("wl_seat", 9)];
        let statuses = required_globals_status(&advertised(&globals));
        assert_eq!(statuses.iter().filter(|s| !s.ok()).count(), 1);
        assert_eq!(
            line_for(&globals, "zcosmic_toplevel_manager_v1"),
            "zcosmic_toplevel_manager_v1: NOT offered (needs v1)",
        );
    }

    #[test]
    fn version_below_the_one_we_bind_is_too_old() {
        let globals = [("wl_seat", 5)];
        assert_eq!(line_for(&globals, "wl_seat"), "wl_seat: TOO OLD (v5, needs v7)");
    }

    #[test]
    fn repeated_global_uses_highest_version() {
        let globals = [("wl_seat", 5), ("wl_seat", 8)];
        assert_eq!(line_for(&globals, "wl_seat"), "wl_seat: ok (v8, needs v7)");
    }
}

def run(argv, **kwargs):
    return subprocess.run([str(x) for x in argv], check=True, **kwargs)


def status():
    return subprocess.run(['systemctl', '--user', 'is-active', '--quiet', SERVICE]).returncode == 0


def atomic_bytes(path, data, mode):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(prefix='.' + path.name + '.', dir=path.parent)
    try:
        with os.fdopen(fd, 'wb') as output:
            output.write(data)
            output.flush()
            os.fsync(output.fileno())
        os.chmod(tmp, mode)
        os.replace(tmp, path)
    finally:
        if os.path.lexists(tmp):
            os.unlink(tmp)


def backup(paths):
    folder = BASE / 'backups' / datetime.datetime.now().strftime('%Y%m%d-%H%M%S-%f')
    folder.mkdir(parents=True, mode=0o700)
    items = []
    for i, target in enumerate(paths):
        item = {'path': str(target), 'kind': 'missing'}
        if target.is_symlink():
            item.update(kind='symlink', link=os.readlink(target))
        elif target.is_file():
            name = str(i) + '.backup'
            shutil.copy2(target, folder / name)
            item.update(kind='file', file=name, mode=target.stat().st_mode & 0o777)
        elif target.exists():
            raise RuntimeError(f'Neocekavany typ cile: {target}')
        items.append(item)
    manifest = {'items': items, 'service_was_active': status()}
    (folder / 'manifest.json').write_text(json.dumps(manifest, indent=2), encoding='utf-8')
    return folder


def restore(folder):
    manifest = json.loads((folder / 'manifest.json').read_text(encoding='utf-8'))
    expected = {str(x) for x in [BIN, UNIT, START, AUTOSTART]}
    if {item['path'] for item in manifest['items']} != expected:
        raise RuntimeError('Zaloha neodpovida teto instalaci.')
    for item in manifest['items']:
        if item['kind'] == 'file' and not (folder / item['file']).is_file():
            raise RuntimeError('V zaloze chybi soubor; obnova nebyla zahajena.')
    subprocess.run(['systemctl', '--user', 'stop', SERVICE], check=False)
    for item in manifest['items']:
        target = Path(item['path'])
        if item['kind'] == 'file':
            atomic_bytes(target, (folder / item['file']).read_bytes(), item['mode'])
        elif item['kind'] == 'symlink':
            target.unlink(missing_ok=True)
            target.symlink_to(item['link'])
        else:
            target.unlink(missing_ok=True)
    run(['systemctl', '--user', 'daemon-reload'])
    if manifest['service_was_active']:
        run(['systemctl', '--user', 'start', SERVICE])
    print('Predchozi instalace byla obnovena.', flush=True)


def import_environment():
    names = [name for name in ['WAYLAND_DISPLAY', 'DISPLAY', 'XDG_CURRENT_DESKTOP', 'XDG_SESSION_TYPE'] if name in os.environ]
    if names:
        run(['systemctl', '--user', 'import-environment', *names])


def desktop_quote(value):
    value = value.replace('%', '%%')
    for character in ['\\', '"', '`', '$']:
        value = value.replace(character, '\\' + character)
    value = value.replace('\\', '\\\\')
    return '"' + value + '"'


def install(source_dir: Path):
    if os.geteuid() == 0:
        raise RuntimeError('Spust instalator jako svuj bezny uzivatel, BEZ sudo.')
    if not os.environ.get('WAYLAND_DISPLAY') or not os.environ.get('XDG_RUNTIME_DIR'):
        raise RuntimeError('Spust instalator v terminalu uvnitr prihlaseneho COSMIC.')
    if not (source_dir / 'Cargo.toml').is_file():
        raise RuntimeError(f'Ve slozce {source_dir} chybi Cargo.toml.')

    cargo = shutil.which('cargo')
    if not cargo and (Path.home() / '.cargo/bin/cargo').is_file():
        cargo = str(Path.home() / '.cargo/bin/cargo')
    if not cargo:
        raise RuntimeError('Chybi cargo/Rust.')

    env = os.environ.copy()
    env['PATH'] = str(Path(cargo).parent) + os.pathsep + env.get('PATH', '')
    env['CARGO_TARGET_DIR'] = str(Path.home() / '.cache/cosmic-alt-tab-mru/target')
    env.setdefault('CARGO_BUILD_JOBS', '4')

    print(f'1/3 Kompiluji a testuji ze slozky: {source_dir}', flush=True)
    run([cargo, 'test', '--locked', '--bin', 'cosmic-ext-app-switcher'], cwd=source_dir, env=env)
    run([cargo, 'build', '--locked', '--release', '--bin', 'cosmic-ext-app-switcher'], cwd=source_dir, env=env)
    new_binary = Path(env['CARGO_TARGET_DIR']) / 'release/cosmic-ext-app-switcher'
    run([new_binary, '--check-compat'])

    print('2/3 Vytvarim zalohu a instaluji prepinac.', flush=True)
    folder = backup([BIN, UNIT, START, AUTOSTART])
    try:
        subprocess.run(['systemctl', '--user', 'stop', SERVICE], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        atomic_bytes(BIN, new_binary.read_bytes(), 0o755)
        unit = '''[Unit]
Description=COSMIC Alt-Tab per-window MRU history
PartOf=graphical-session.target
After=graphical-session.target

[Service]
Type=simple
ExecStart="%h/.local/bin/cosmic-ext-app-switcher" --daemon
Restart=on-failure
RestartSec=2
UMask=0077
'''
        atomic_bytes(UNIT, unit.encode(), 0o644)
        starter = '''#!/bin/sh
set -eu
[ -n "${WAYLAND_DISPLAY:-}" ] || exit 0
systemctl --user import-environment WAYLAND_DISPLAY
[ -z "${DISPLAY:-}" ] || systemctl --user import-environment DISPLAY
systemctl --user restart cosmic-ext-app-switcher-mru.service
'''
        atomic_bytes(START, starter.encode(), 0o755)
        desktop = '[Desktop Entry]\nType=Application\nName=COSMIC Alt-Tab MRU history\nExec=' + desktop_quote(str(START)) + '\nTerminal=false\nOnlyShowIn=COSMIC;\n'
        atomic_bytes(AUTOSTART, desktop.encode(), 0o644)
        run(['systemctl', '--user', 'daemon-reload'])
        import_environment()
        run(['systemctl', '--user', 'start', SERVICE])

        print('3/3 Kontroluji spojeni s MRU sluzbou.', flush=True)
        deadline = time.monotonic() + 15
        while True:
            result = subprocess.run([BIN, '--mru-check'], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            if result.returncode == 0:
                break
            if time.monotonic() >= deadline:
                raise RuntimeError('Sluzba se nespustila; obnovuji puvodni instalaci.')
            time.sleep(0.2)
        atomic_bytes(BASE / 'last-backup', (str(folder) + '\n').encode(), 0o600)
    except BaseException:
        restore(folder)
        raise

    print('\nHOTOVO. Prepinac je nainstalovany a sluzba bezi.')


def main():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument('--rollback', action='store_true', help='Obnovi predchozi zalohu')
    parser.add_argument('--source', type=Path, default=PROJECT_DIR, help='Cesta ke slozce s Cargo.toml')
    args = parser.parse_args()

    BASE.mkdir(parents=True, exist_ok=True)
    with (BASE / 'installer.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        if args.rollback:
            marker = BASE / 'last-backup'
            if not marker.is_file():
                raise RuntimeError('Nebyla nalezena zaloha z uspesne instalace.')
            restore(Path(marker.read_text().strip()))
            marker.unlink()
        else:
            install(args.source.resolve())


if __name__ == '__main__':
    try:
        main()
    except (Exception, KeyboardInterrupt) as error:
        print('\nChyba:', error, file=sys.stderr)
        raise SystemExit(1)
