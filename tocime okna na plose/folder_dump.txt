# Rychlý přepínač oken na aktuální ploše pro COSMIC (`cosmic-ws-switcher` v0.2.0)

Tento nástroj přidává do prostředí **COSMIC Desktop (Pop!_OS)** bleskové cyklické přepínání oken pouze v rámci právě aktivní pracovní plochy (workspace) pomocí klávesových zkratek **`Super + ←`** (předchozí okno) a **`Super + →`** (další okno).

## Proč vzniklo toto řešení a jak funguje
Výchozí přepínání nebo volání jednorázových příkazů přes CLI nástroj `cosmicmsg` trvá při každém stisku klávesy přibližně **300 ms** (kvůli opakovanému navazování spojení s Waylandem a synchronizaci stavů). Navíc knihovní funkce `cosmicmsg::subscribe` trpí chybou při sledování ploch a ve vícevláknovém režimu způsobuje přeskakování fokusu zpět na staré okno (race condition).

**Architektura `v0.2.0` tyto problémy kompletně odstraňuje:**
* **Jediné trvalé Wayland spojení:** Daemon běží v jednom vlákně nad `libc::poll` a současně sleduje Wayland socket i lokální Unix socket (`$XDG_RUNTIME_DIR/cosmic-ws-switcher.sock`).
* **Okamžitá odezva (~0,15 až 0,25 ms):** Stav všech oken a ploch je neustále udržován v RAM (spotřeba ~1 MB). Při stisku zkratky se pouze odešle přímý Wayland požadavek `activate` + `flush()`.
* **Stabilní pořadí a Optimistic Lock:** Okna se řadí deterministicky podle pořadí otevření a po dobu 300 ms od stisku platí optimistický zámek, takže při zběsilém mačkání šipek nikdy nedojde k vrácení na předchozí okno.

use cosmicmsg::{collect_state, window_activate, AppData, EventQueue};
use std::{
    collections::HashSet,
    env, fs,
    io::{Read, Write},
    os::fd::{AsFd, AsRawFd},
    os::unix::net::{UnixListener, UnixStream},
    path::PathBuf,
    time::{Duration, Instant},
};

#[derive(Debug, Clone, Copy)]
enum Direction {
    Next,
    Prev,
}

struct Tracker {
    /// Stabilní pořadí oken podle toho, kdy vznikla (aby rotace nepřeskakovala tam a zpět)
    order: Vec<String>,
    /// Poslední známé aktivní okno z Waylandu
    wayland_active: Option<String>,
    /// Optimisticky vybrané okno při rychlém mačkání kláves (ochrana před race condition)
    optimistic_target: Option<(String, Instant)>,
    /// Statistiky posledního přepnutí v mikrosekundách
    last_select_us: u128,
    last_activate_us: u128,
    total_switches: u64,
}

impl Tracker {
    fn new() -> Self {
        Self {
            order: Vec::new(),
            wayland_active: None,
            optimistic_target: None,
            last_select_us: 0,
            last_activate_us: 0,
            total_switches: 0,
        }
    }

    /// Synchronizuje naše stabilní pole `order` a `wayland_active` přímo z živého `AppData`
    fn sync_from_wayland(&mut self, app_data: &AppData) {
        let toplevels = app_data.toplevels();
        let mut current_ids = HashSet::with_capacity(toplevels.len());
        let mut newly_activated: Option<String> = None;

        for win in &toplevels {
            current_ids.insert(win.identifier.clone());
            if !self.order.contains(&win.identifier) {
                self.order.push(win.identifier.clone());
            }
            if win.state.iter().any(|s| s == "activated") {
                newly_activated = Some(win.identifier.clone());
            }
        }

        // Odstranit zavřená okna ze seznamu pořadí
        self.order.retain(|id| current_ids.contains(id));

        if let Some(act) = newly_activated {
            self.wayland_active = Some(act.clone());
            if let Some((ref opt_id, _)) = self.optimistic_target {
                if opt_id == &act {
                    self.optimistic_target = None;
                }
            }
        }
    }

    /// Zjistí, které okno považujeme v tento okamžik za výchozí pro další krok rotace
    fn effective_current(&mut self) -> Option<String> {
        if let Some((ref target, ts)) = self.optimistic_target {
            if ts.elapsed() < Duration::from_millis(300) && self.order.contains(target) {
                return Some(target.clone());
            }
            self.optimistic_target = None;
        }
        self.wayland_active.clone()
    }

    fn select(&mut self, app_data: &AppData, direction: Direction) -> Option<String> {
        self.sync_from_wayland(app_data);

        let workspaces = app_data.workspaces();
        let active_ws_names: HashSet<String> = workspaces
            .into_iter()
            .filter(|ws| ws.active)
            .map(|ws| ws.name)
            .collect();

        let toplevels = app_data.toplevels();
        let cur_id = self.effective_current();

        let target_ws = cur_id
            .as_ref()
            .and_then(|cid| toplevels.iter().find(|t| &t.identifier == cid))
            .and_then(|win| {
                win.workspaces
                    .iter()
                    .find(|ws| active_ws_names.contains(*ws))
                    .or_else(|| win.workspaces.first())
                    .cloned()
            })
            .or_else(|| active_ws_names.iter().next().cloned())?;

        let candidates: Vec<String> = self
            .order
            .iter()
            .filter(|id| {
                if let Some(win) = toplevels.iter().find(|t| &t.identifier == *id) {
                    let is_sticky = win.state.iter().any(|s| s == "sticky");
                    let on_ws = win.workspaces.iter().any(|ws| ws == &target_ws);
                    is_sticky || on_ws
                } else {
                    false
                }
            })
            .cloned()
            .collect();

        if candidates.len() <= 1 {
            return None;
        }

        let current_index = cur_id
            .as_ref()
            .and_then(|cid| candidates.iter().position(|id| id == cid));

        let next_index = match (direction, current_index) {
            (Direction::Next, Some(i)) => (i + 1) % candidates.len(),
            (Direction::Prev, Some(i)) => {
                if i == 0 {
                    candidates.len() - 1
                } else {
                    i - 1
                }
            }
            (Direction::Next, None) => 0,
            (Direction::Prev, None) => candidates.len() - 1,
        };

        let chosen = candidates[next_index].clone();
        self.optimistic_target = Some((chosen.clone(), Instant::now()));
        Some(chosen)
    }
}

fn socket_path() -> PathBuf {
    if let Ok(runtime) = env::var("XDG_RUNTIME_DIR") {
        PathBuf::from(runtime).join("cosmic-ws-switcher.sock")
    } else {
        PathBuf::from(format!("/tmp/cosmic-ws-switcher-{}.sock", unsafe {
            libc::getuid()
        }))
    }
}

/// Přečte všechny čekající události z Wayland socketu bez blokování
fn pump_wayland(
    app_data: &mut AppData,
    eq: &mut EventQueue<AppData>,
    tracker: &mut Tracker,
) -> Result<(), String> {
    let _ = eq.flush();
    if let Some(guard) = eq.prepare_read() {
        match guard.read() {
            Ok(_) => {}
            Err(wayland_client::backend::WaylandError::Io(ref e))
                if e.kind() == std::io::ErrorKind::WouldBlock => {}
            Err(e) => return Err(format!("Wayland read error: {e}")),
        }
    }
    eq.dispatch_pending(app_data)
        .map_err(|e| format!("Wayland dispatch error: {e}"))?;
    tracker.sync_from_wayland(app_data);
    Ok(())
}

fn handle_client(
    mut stream: UnixStream,
    app_data: &mut AppData,
    eq: &mut EventQueue<AppData>,
    tracker: &mut Tracker,
) {
    let mut buf = [0u8; 64];
    let n = match stream.read(&mut buf) {
        Ok(n) if n > 0 => n,
        _ => return,
    };
    let cmd = std::str::from_utf8(&buf[..n]).unwrap_or("").trim();

    match cmd {
        "next" | "prev" => {
            let dir = if cmd == "next" {
                Direction::Next
            } else {
                Direction::Prev
            };

            let t0 = Instant::now();
            let _ = pump_wayland(app_data, eq, tracker);
            let target = tracker.select(app_data, dir);
            let select_us = t0.elapsed().as_micros();

            if let Some(identifier) = target {
                let t1 = Instant::now();
                if let Err(e) = window_activate(app_data, eq, &identifier) {
                    eprintln!("Chyba pri aktivaci okna {identifier}: {e}");
                }
                let activate_us = t1.elapsed().as_micros();

                tracker.last_select_us = select_us;
                tracker.last_activate_us = activate_us;
                tracker.total_switches += 1;

                let _ = writeln!(
                    stream,
                    "ok target={} select_us={} activate_us={}",
                    identifier, select_us, activate_us
                );
            } else {
                let _ = writeln!(stream, "noop select_us={}", select_us);
            }
        }
        "ping" => {
            let _ = stream.write_all(b"pong\n");
        }
        "status" => {
            let _ = pump_wayland(app_data, eq, tracker);
            let toplevels = app_data.toplevels();
            let active_ws: Vec<String> = app_data
                .workspaces()
                .into_iter()
                .filter(|w| w.active)
                .map(|w| w.name)
                .collect();

            let _ = writeln!(
                stream,
                "windows={} wayland_active={:?} optimistic={:?} active_ws={:?} last_select_us={} last_activate_us={} total_switches={}",
                toplevels.len(),
                tracker.wayland_active,
                tracker.optimistic_target.as_ref().map(|(id, _)| id),
                active_ws,
                tracker.last_select_us,
                tracker.last_activate_us,
                tracker.total_switches
            );
            for (idx, id) in tracker.order.iter().enumerate() {
                if let Some(w) = toplevels.iter().find(|t| &t.identifier == id) {
                    let _ = writeln!(
                        stream,
                        "  [{}] id={} app={} ws={:?} state={:?} title={:?}",
                        idx, w.identifier, w.app_id, w.workspaces, w.state, w.title
                    );
                }
            }
        }
        other => {
            let _ = writeln!(stream, "err unknown command: {other}");
        }
    }
}

fn run_daemon() -> Result<(), Box<dyn std::error::Error>> {
    let sock = socket_path();
    if sock.exists() {
        let _ = fs::remove_file(&sock);
    }

    let (mut app_data, mut eq) = collect_state()?;
    let mut tracker = Tracker::new();
    tracker.sync_from_wayland(&app_data);

    let listener = UnixListener::bind(&sock)?;
    listener.set_nonblocking(true)?;

    let wl_fd = eq.as_fd().as_raw_fd();
    let sock_fd = listener.as_raw_fd();

    eprintln!(
        "cosmic-ws-switcher v0.2.0 daemon bezi na {} (oken: {})",
        sock.display(),
        tracker.order.len()
    );

    loop {
        let _ = eq.flush();

        let mut fds = [
            libc::pollfd {
                fd: wl_fd,
                events: libc::POLLIN,
                revents: 0,
            },
            libc::pollfd {
                fd: sock_fd,
                events: libc::POLLIN,
                revents: 0,
            },
        ];

        let ret = unsafe { libc::poll(fds.as_mut_ptr(), fds.len() as libc::nfds_t, -1) };
        if ret < 0 {
            let err = std::io::Error::last_os_error();
            if err.kind() == std::io::ErrorKind::Interrupted {
                continue;
            }
            return Err(Box::new(err));
        }

        if (fds[0].revents & (libc::POLLIN | libc::POLLERR | libc::POLLHUP)) != 0 {
            if let Err(e) = pump_wayland(&mut app_data, &mut eq, &mut tracker) {
                eprintln!("{e}");
                break;
            }
        }

        if (fds[1].revents & libc::POLLIN) != 0 {
            loop {
                match listener.accept() {
                    Ok((stream, _)) => {
                        handle_client(stream, &mut app_data, &mut eq, &mut tracker);
                    }
                    Err(ref e) if e.kind() == std::io::ErrorKind::WouldBlock => break,
                    Err(e) => {
                        eprintln!("Socket accept error: {e}");
                        break;
                    }
                }
            }
        }
    }

    Ok(())
}

fn send_command(command: &str, print_reply: bool) -> Result<(), Box<dyn std::error::Error>> {
    let sock = socket_path();
    let mut stream = UnixStream::connect(&sock)?;
    stream.write_all(command.as_bytes())?;
    stream.shutdown(std::net::Shutdown::Write)?;

    if print_reply {
        let mut response = String::new();
        stream.read_to_string(&mut response)?;
        print!("{response}");
    }
    Ok(())
}

fn run_bench() -> Result<(), Box<dyn std::error::Error>> {
    println!("=== Interni benchmark cosmic-ws-switcher v0.2.0 ===");
    send_command("status", true)?;
    println!("\nTestuji 10x prepnuti (5x next, 5x prev) s pauzou 80 ms...");

    for i in 1..=10 {
        let cmd = if i <= 5 { "next" } else { "prev" };
        let t0 = Instant::now();
        let sock = socket_path();
        let mut stream = UnixStream::connect(&sock)?;
        stream.write_all(cmd.as_bytes())?;
        stream.shutdown(std::net::Shutdown::Write)?;

        let mut reply = String::new();
        stream.read_to_string(&mut reply)?;
        let total_us = t0.elapsed().as_micros();

        println!(
            "Krok {:2} ({}): celkem IPC+Wayland = {:4} µs ({:.2} ms) | {}",
            i,
            cmd,
            total_us,
            total_us as f64 / 1000.0,
            reply.trim()
        );
        std::thread::sleep(Duration::from_millis(80));
    }

    println!("\nStav po testu:");
    send_command("status", true)?;
    Ok(())
}

fn main() {
    let arg = env::args().nth(1).unwrap_or_else(|| "help".to_string());

    let result = match arg.as_str() {
        "daemon" => run_daemon(),
        "next" => send_command("next", false),
        "prev" => send_command("prev", false),
        "ping" => send_command("ping", true),

            );
            return;
        }
    };

    if let Err(e) = result {
        eprintln!("cosmic-ws-switcher: {e}");
        std::process::exit(1);
    }
}

-- Samostatný rychlý přesouvač oken mezi plochami (Title Bar Hold + Ctrl+Arrow).
-- Obsahuje živý přepínač (Option+Ctrl+Shift+M) mezi rychlou a původní Mission Control verzí.

local M = { busy = false, hotkeys = {}, toggleHotkey = nil, enabled = false }
local events = hs.eventtap.event
local mouseAPI = hs.mouse
local timers = {}

local function clearTimers()
    for _, t in ipairs(timers) do t:stop() end
    timers = {}
end

local function later(seconds, fn)
    timers[#timers + 1] = hs.timer.doAfter(seconds, fn)
end

local function postMouse(kind, pt)
    events.newMouseEvent(events.types[kind], pt):post()
end

local function getDragPoint(win)
    local f = win:frame()
    local xOffset = math.min(math.max(math.floor(f.w * 0.35), 90), f.w - 20)
    return { x = f.x + xOffset, y = f.y + 14 }
end

function M.move(win, target)
    if M.busy then return false, 'busy' end
    if not win or not win:isStandard() or win:isFullScreen() then
        hs.alert.show('Vyber běžné okno mimo celou obrazovku.')
        return false
    end

    local screen = win:screen()
    local spaces = hs.spaces.spacesForScreen(screen)
    local current = hs.spaces.activeSpaceOnScreen(screen)
    if current == target then return true end

    local curIdx, targetIdx
    for i, id in ipairs(spaces) do
        if id == current then curIdx = i end
        if id == target then targetIdx = i end
    end
    if not curIdx or not targetIdx then return false end

    local diff = targetIdx - curIdx
    local keyDir = diff < 0 and 'left' or 'right'
    local steps = math.abs(diff)

    M.busy = true
    local started = hs.timer.absoluteTime() / 1e9
    local origMouse = mouseAPI.absolutePosition()
    local dragPt = getDragPoint(win)

    local waitStart = hs.timer.absoluteTime() / 1e9
    local function waitForModifiersRelease()
        local mods = hs.eventtap.checkKeyboardModifiers()
        if mods.alt or mods.shift or mods.cmd then
            if (hs.timer.absoluteTime() / 1e9) - waitStart > 1.5 then
                M.busy = false
                hs.alert.show('Při přesunu pusť klávesy Option/Shift.')
                return
            end
            later(0.01, waitForModifiersRelease)
            return
        end

        win:focus()
        postMouse('mouseMoved', dragPt)
        postMouse('leftMouseDown', dragPt)

        later(0.02, function()
            postMouse('leftMouseDragged', { x = dragPt.x + 1, y = dragPt.y })

            for s = 1, steps do
                later(0.02 + (s - 1) * 0.12, function()
                    events.newKeyEvent({'ctrl'}, keyDir, true):post()
                    events.newKeyEvent({'ctrl'}, keyDir, false):post()
                end)
            end

            local releaseDelay = 0.08 + steps * 0.12
            later(releaseDelay, function()
                postMouse('leftMouseDragged', dragPt)
                postMouse('leftMouseUp', dragPt)
                mouseAPI.absolutePosition(origMouse)
                win:focus()
                M.busy = false
                M.lastResult = {
                    ok = true,
                    elapsed = (hs.timer.absoluteTime() / 1e9) - started
                }
                hs.printf('Fast space move: PASS, %.2f s', M.lastResult.elapsed)
            end)
        end)

function M.moveDirection(win, dir)
    if M.busy then return false, 'busy' end
    win = win or hs.window.focusedWindow()
    if not win or not win:isStandard() or win:isFullScreen() then
        hs.alert.show('Vyber běžné okno mimo celou obrazovku.')
        return false
    end

    local keyDir = dir
    if dir == 'up' then keyDir = 'left'
    elseif dir == 'down' then keyDir = 'right' end

    local screen = win:screen()
    local spaces = hs.spaces.spacesForScreen(screen)
    local current = hs.spaces.activeSpaceOnScreen(screen)
    local idx
    for i, id in ipairs(spaces) do
        if id == current then idx = i; break end

    if not idx then return false end

    local targetIdx = idx + (keyDir == 'left' and -1 or 1)
    if targetIdx < 1 or targetIdx > #spaces then
        return false, 'edge'
    end

    return M.move(win, spaces[targetIdx])
end

    local id,screen,frame=win:id(),win:screen(),win:frame()
    local area=screen:frame()

# Ovládání macOS po vzoru Pop!_OS

Aktivní profil pro Hammerspoon. **Option nahrazuje Super**.
Načítá se přes `/Users/jirka/.hammerspoon/init.lua`; Hammerspoon se spouští při přihlášení.

| Zkratka | Akce |
|---|---|
| Option + → / ↓ | Další okno, po posledním znovu první |
| Option + ← / ↑ | Předchozí okno, po prvním znovu poslední |
| Option + Shift + šipka | Rozložení aktivního volného okna směrem k okraji |
| Option + Control + → / ↓ | Další běžná plocha |
| Option + Control + ← / ↑ | Předchozí běžná plocha |
| Option + Control + Shift + šipka | Přesun okna na sousední plochu a přechod s ním |
| Option + W | Otevřít / zavřít Mission Control |
| Option + Tab | Dosavadní AltTab |

## Rozkládání oken

Používá přechody volných oken z COSMIC, nikoli automatické dlaždicování všech oken:

- Ze základní velikosti: vlevo, vpravo, nahoru nebo dolů → příslušná polovina.
- Kolmá šipka → roh. Například vlevo, pak nahoru → levá horní čtvrtina.
- Opačná šipka rozšíří danou osu. Levý horní roh + dolů → levá polovina.
- Levá polovina + doprava → maximalizace v použitelné ploše obrazovky.
- Opakovaná šipka ven z již dosaženého okraje okno nechá na místě.
  Přesun na další plochu je záměrně na výslovné kombinaci s Control.
  COSMIC může v tomto bodě přesouvat dál na plochu nebo monitor; tato část není shodná.
- Ruční změna velikosti/polohy se rozpozná; další šipka vychází z aktuálního okna.
- Aplikace si může vynutit minimální velikost. Například zdejší Finder nedovolí
  šířku menší než 843 bodů, zatímco polovina monitoru má 756. Takové okno zůstane
  přisazené ke správnému okraji, ale bude širší než polovina.

## Přesun mezi plochami

Přesun probíhá přes náhledy v Mission Control. Běžné rozhraní Hammerspoonu na tomto
Macu vracelo úspěch, ale okno jiné aplikace se ve skutečnosti nepřesunulo.
Nový postup skutečné umístění, aktivní plochu a fokus kontroluje.

Po stisku **pusť modifikátory**. Po zrychlení 16. 10. trval celý přesun v opakovaném testu **1,95–2,25 sekundy**
(původní verze přibližně 3–5 sekund);
na chvíli se zobrazí Mission Control a pohne kurzor. Po dokončení se kurzor vrátí.
Během přesunu nehýbej myší. Escape přesun zruší. Další naše zkratky během
přesunu nevstupují do rozpracované operace.

- ← a ↑ znamenají předchozí, → a ↓ následující plochu v aktuálním pořadí macOS.
- Na prvním/posledním místě se nepřeskakuje na opačný konec a nevytvářejí se nové plochy.

- Mission Control neposkytuje ID náhledů oken. Skript proto vyžaduje jediný přesně

-- Geometry-only floating-window transitions: halves, corners, opposite expands.
local M = {}
local directions = {left={'x',-1},right={'x',1},up={'y',-1},down={'y',1}}
function M.next(state, direction)
    local d=assert(directions[direction], 'invalid direction')
    local nextState={x=state and state.x or 0, y=state and state.y or 0}
    if nextState[d[1]]==d[2] then return nextState, true end
    nextState[d[1]]=nextState[d[1]]==-d[2] and 0 or d[2]
    return nextState, false
end
function M.frame(state, area)
    local w=state.x==0 and area.w or math.floor(area.w/2)
    local h=state.y==0 and area.h or math.floor(area.h/2)
    return {x=area.x+(state.x==1 and area.w-w or 0),
            y=area.y+(state.y==1 and area.h-h or 0),w=w,h=h}
end
function M.near(a,b)
    for _,k in ipairs({'x','y','w','h'}) do if math.abs(a[k]-b[k])>3 then return false end end
    return true
end
function M.infer(frame,area)
    for x=-1,1 do for y=-1,1 do
        local state={x=x,y=y}
        if M.near(frame,M.frame(state,area)) then return state end
    end end
end
return M

-- Move ordinary windows through Mission Control on macOS 26.

local M = {busy=false}

    lastPoint=point

end

    for _,e in ipairs(parent and parent.AXChildren or {}) do
        if e.AXIdentifier==identifier then return e end
    end
end

end

    local mc=missionControl()

        if e.AXIdentifier=='mc.display' and e.AXDisplayID==screen:id() then return e end
    end
end
local function finish(ok,message)
    for _,t in ipairs(timers) do t:stop() end
    timers={}
    if mouseDown then mouse('leftMouseUp',lastPoint); mouseDown=false end
    if escape then escape:delete(); escape=nil end
    if not ok then hs.spaces.closeMissionControl() end
    if pointer then mouseAPI.absolutePosition(pointer); pointer=nil end
    M.busy=false
    M.lastResult={ok=ok,message=message,elapsed=now()-started,phases=phases,readiness=M.readiness}
    hs.printf('Space move: %s, %.2f s',ok and 'PASS' or 'FAIL',M.lastResult.elapsed)
    if not ok and message then hs.alert.show(message) end
end
local function later(seconds,fn)
    timers[#timers+1]=hs.timer.doAfter(seconds,function()
        local ok,err=xpcall(fn,debug.traceback)
        if not ok then hs.printf('Space move: %s',err); finish(false,'Přesun okna selhal.') end
    end)
end
local function waitUntil(test,success,timeout,message)
    local deadline=now()+timeout
    local function check()
        local result=test()
        if result then success(result)
        elseif now()>=deadline then finish(false,message)
        else later(0.05,check) end
    end
    later(0.05,check)
end
local function center(f) return {x=f.x+f.w/2,y=f.y+f.h/2} end
local function near(a,b)
    if not a or not b then return false end
    for _,k in ipairs({'x','y','w','h'}) do if math.abs(a[k]-b[k])>1 then return false end end
    return true
end

    return false

function M.move(win,target)
    if M.busy then return false,'busy' end
    if not win or not win:isStandard() or win:isFullScreen() then
        hs.alert.show('Vyber běžné okno mimo celou obrazovku.'); return false
    end
    local screen=win:screen()
    local spaces=hs.spaces.windowSpaces(win) or {}
    if #spaces~=1 or hs.spaces.spaceType(spaces[1])~='user' then
        hs.alert.show('Okno musí patřit jedné běžné ploše.'); return false
    end
    if hs.spaces.spaceType(target)~='user' or hs.spaces.spaceDisplay(target)~=screen:getUUID() then return false end
    if onSpace(win,target) then return true end

    for i,id in ipairs(hs.spaces.spacesForScreen(screen)) do if id==target then index=i end end
    if not index then return false end

local function candidates(screenID)

                ranks[id] = nextRank
            end
            result[#result + 1] = win
        end
    end
    -- Retain our own ranks: equal creation timestamps must not reorder windows.
    table.sort(result, function(a, b) return ranks[a:id()] < ranks[b:id()] end)
    return result
end

local function cycle(step)
    if not filter or (enabledWhen and not enabledWhen()) then return end
    local focused = hs.window.focusedWindow()
    local screen = (focused and focused:screen()) or hs.screen.mainScreen()
    local screenID = screen and screen:id()
    local windows = candidates(screenID)
    if #windows <= 1 then resetPending(); return end

    local now = hs.timer.absoluteTime() / 1e9
    local currentID = focused and focused:id()
    if pendingID and now < pendingUntil and pendingScreen == screenID then
        -- AX focus can arrive after another keypress. Advance from our target.
        for _, win in ipairs(windows) do
            if win:id() == pendingID then currentID = pendingID; break end
        end
    end

    local index
    for i, win in ipairs(windows) do
        if win:id() == currentID then index = i; break end
    end
    local targetIndex
    if index then
        targetIndex = ((index - 1 + step) % #windows) + 1
    else
        targetIndex = step > 0 and 1 or #windows
    end

    local target = windows[targetIndex]
    pendingID, pendingUntil, pendingScreen = target:id(), now + 0.8, screenID
    local ok, result = pcall(function() return target:focus() end)
    if not ok or not result then
        resetPending()
        hs.printf("Window cycle: could not focus window %s", tostring(target:id()))
        return nil
    end
    return target:id()
end

function M.next() return cycle(1) end
function M.previous() return cycle(-1) end

function M.stop()
    for _, key in ipairs(hotkeys) do key:delete() end
    hotkeys = {}
    if spaceWatcher then spaceWatcher:stop(); spaceWatcher = nil end
    if filter then filter:unsubscribeAll():pause(); filter = nil end
    ranks, nextRank = {}, 0
    resetPending()
    return M
end

function M.start(options)
    M.stop()
    options = options or {}
    onlyCurrentScreen = options.onlyCurrentScreen ~= false
    enabledWhen = options.enabledWhen
    local mods = options.modifiers or {"ctrl", "alt"}
    filter = hs.window.filter.new():setOverrideFilter({
        currentSpace = true,
        visible = true,
        -- Auxiliary dialogs may activate their parent instead of themselves.
        -- Including one would send the cycle back to the parent's position.
        allowRoles = {"AXStandardWindow"},
    })
    filter:subscribe(hs.window.filter.windowDestroyed, function(win)
        local id = win:id()
        if id then ranks[id] = nil end
        if id == pendingID then resetPending() end
    end)
    spaceWatcher = hs.spaces.watcher.new(resetPending):start()

    local function bind(key, callback)
        local binding = hs.hotkey.bind(mods, key, callback, nil, callback)
        if not binding then
            M.stop()
            error("Window cycle: cannot register shortcut " .. key)
        end
        hotkeys[#hotkeys + 1] = binding
    end
    bind("right", M.next)
    bind("left", M.previous)
    return M
end

return M
